package sipinjam.aturan;

import java.time.temporal.ChronoUnit;

import sipinjam.exception.AturanDilanggarException;

/** R4: barang paling lambat H-1, ruangan paling lambat H-3. */
public class AturanBatasWaktu implements AturanPeminjaman {
    private static final int MINIMAL_BARANG = 1;
    private static final int MINIMAL_RUANGAN = 3;

    @Override
    public String getNamaAturan() {
        return "R4 Batas waktu pengajuan";
    }

    @Override
    public void periksa(KonteksPengajuan konteks) throws AturanDilanggarException {
        long selisih = ChronoUnit.DAYS.between(konteks.getHariIni(), konteks.getTanggalMulai());
        int minimal = konteks.isRuangan() ? MINIMAL_RUANGAN : MINIMAL_BARANG;
        if (selisih < minimal) {
            throw new AturanDilanggarException(getNamaAturan(),
                    "Pengajuan paling lambat H-" + minimal + ", sedangkan jarak saat ini H-" + selisih);
        }
    }
}

package sipinjam.aturan;

import sipinjam.exception.AturanDilanggarException;
import sipinjam.exception.BentrokJadwalException;
import sipinjam.model.BookingRuangan;

/** S3: menolak slot yang bertumpang tindih dengan booking aktif lain. */
public class AturanBentrokJadwal implements AturanPeminjaman {

    @Override
    public String getNamaAturan() {
        return "S3 Bentrok jadwal";
    }

    @Override
    public void periksa(KonteksPengajuan konteks) throws AturanDilanggarException {
        if (!konteks.isRuangan()) {
            return;
        }
        BookingRuangan bentrok = konteks.getRepoBooking().cariBentrok(
                konteks.getRuang(), konteks.getTanggalMulai(), konteks.getSlot());
        if (bentrok != null) {
            throw new BentrokJadwalException("Slot " + konteks.getSlot() + " pada "
                    + konteks.getRuang().getNama() + " bentrok dengan booking " + bentrok.getSlot()
                    + " milik " + bentrok.getPemohon().getNama()
                    + " (" + bentrok.getStatus().getLabel() + ")");
        }
    }
}

package sipinjam.aturan;

import sipinjam.exception.AturanDilanggarException;

/** R5: barang maksimal tujuh hari, ruangan minimal satu jam dan maksimal empat jam. */
public class AturanDurasiMaksimal implements AturanPeminjaman {
    private static final int MAKS_HARI = 7;
    private static final int MAKS_JAM = 4;
    private static final int MIN_JAM = 1;

    @Override
    public String getNamaAturan() {
        return "R5 Durasi maksimal";
    }

    @Override
    public void periksa(KonteksPengajuan konteks) throws AturanDilanggarException {
        if (konteks.isRuangan()) {
            int jam = konteks.getSlot().durasiJam();
            if (jam < MIN_JAM || jam > MAKS_JAM) {
                throw new AturanDilanggarException(getNamaAturan(),
                        "Durasi ruangan harus " + MIN_JAM + " sampai " + MAKS_JAM + " jam, bukan " + jam + " jam");
            }
        } else if (konteks.getDurasiHari() > MAKS_HARI) {
            throw new AturanDilanggarException(getNamaAturan(),
                    "Durasi peminjaman maksimal " + MAKS_HARI + " hari, bukan " + konteks.getDurasiHari() + " hari");
        }
    }
}

package sipinjam.aturan;

import java.time.LocalTime;

import sipinjam.exception.AturanDilanggarException;
import sipinjam.exception.DiLuarJamOperasionalException;

/** S1 dan S2: slot berada di antara 07.00-21.00 dan kelipatan satu jam. */
public class AturanJamOperasional implements AturanPeminjaman {
    private static final LocalTime BUKA = LocalTime.of(7, 0);
    private static final LocalTime TUTUP = LocalTime.of(21, 0);

    @Override
    public String getNamaAturan() {
        return "S1 Jam operasional";
    }

    @Override
    public void periksa(KonteksPengajuan konteks) throws AturanDilanggarException {
        if (!konteks.isRuangan()) {
            return;
        }
        if (!konteks.getSlot().diDalam(BUKA, TUTUP)) {
            throw new DiLuarJamOperasionalException("Slot " + konteks.getSlot()
                    + " berada di luar jam operasional 07.00-21.00");
        }
        if (!konteks.getSlot().kelipatanJam()) {
            throw new DiLuarJamOperasionalException("Slot " + konteks.getSlot()
                    + " bukan kelipatan satu jam");
        }
    }
}

package sipinjam.aturan;

/** S4: jumlah peserta tidak boleh melebihi kapasitas ruangan. */
public class AturanKapasitasRuang implements AturanPeminjaman {

    @Override
    public String getNamaAturan() {
        return "S4 Kapasitas ruangan";
    }

    @Override

                    + " orang melebihi kapasitas " + konteks.getRuang().getNama()
                    + " (" + konteks.getRuang().getKapasitas() + " orang)");
        }
    }
}

package sipinjam.aturan;

import sipinjam.exception.AturanDilanggarException;
import sipinjam.exception.StokTidakCukupException;
import sipinjam.model.DetailPeminjaman;

/** Pemeriksaan stok barang pada rentang tanggal yang diminta. */
public class AturanKetersediaanStok implements AturanPeminjaman {

    @Override
    public String getNamaAturan() {
        return "Ketersediaan stok";
    }

    @Override
    public void periksa(KonteksPengajuan konteks) throws AturanDilanggarException {
        if (konteks.isRuangan()) {
            return;
        }
        for (DetailPeminjaman baris : konteks.getDetail()) {
            int terpakai = konteks.getRepoPeminjaman().terpakai(baris.getBarang(),
                    konteks.getTanggalMulai(), konteks.getTanggalSelesai());
            int sisa = baris.getBarang().getStok() - terpakai;
            if (baris.getJumlah() > sisa) {
                throw new StokTidakCukupException("Stok " + baris.getBarang().getNama()
                        + " tersisa " + Math.max(0, sisa) + " unit, diminta " + baris.getJumlah() + " unit");
            }
        }
    }
}

/**
 * Data yang dibutuhkan seluruh aturan dalam satu pemeriksaan.
 * Dua pembuat objek (untuk barang dan untuk ruangan) memakai overloading.
 */
public class KonteksPengajuan {
    private final Mahasiswa pemohon;
    private final boolean ruangan;
    private final LocalDate tanggalMulai;
    private final LocalDate tanggalSelesai;
    private final int durasiHari;
    private final SlotWaktu slot;
    private final Ruangan ruang;

    private final List<DetailPeminjaman> detail;
    private final RepositoriPeminjaman repoPeminjaman;
    private final RepositoriBooking repoBooking;
    private final LocalDate hariIni;

    private KonteksPengajuan(Mahasiswa pemohon, boolean ruangan, LocalDate tanggalMulai,
                             LocalDate tanggalSelesai, int durasiHari, SlotWaktu slot,
                             Ruangan ruang, int jumlahPeserta,
                             List<DetailPeminjaman> detail, RepositoriPeminjaman repoPeminjaman,
                             RepositoriBooking repoBooking, LocalDate hariIni) {

        this.ruangan = ruangan;
        this.tanggalMulai = tanggalMulai;
        this.tanggalSelesai = tanggalSelesai;
        this.durasiHari = durasiHari;
        this.slot = slot;
        this.ruang = ruang;
        this.jumlahPeserta = jumlahPeserta;
        this.detail = detail;
        this.repoPeminjaman = repoPeminjaman;

                                               List<DetailPeminjaman> detail,
                                               RepositoriPeminjaman repoPeminjaman,
                                               RepositoriBooking repoBooking, LocalDate hariIni) {
        int durasi = (int) Math.max(1, java.time.temporal.ChronoUnit.DAYS.between(mulai, selesai));
        return new KonteksPengajuan(pemohon, false, mulai, selesai, durasi, null, null, 0, detail,

    }

    public static KonteksPengajuan untukRuangan(Mahasiswa pemohon, Ruangan ruang, LocalDate tanggal,
                                                SlotWaktu slot, int jumlahPeserta,

package sipinjam.aturan;

import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.List;

import sipinjam.exception.AturanDilanggarException;

/**
 * Menjalankan seluruh aturan yang terdaftar pada satu koleksi.
 * Penambahan aturan baru cukup dengan menambah objek pada daftar.
 */
public class ValidatorPeminjaman {
    private final List<AturanPeminjaman> daftarAturan;

    public ValidatorPeminjaman(List<AturanPeminjaman> daftarAturan) {
        this.daftarAturan = new ArrayList<AturanPeminjaman>(daftarAturan);
    }

    public ValidatorPeminjaman() {
        this(Arrays.<AturanPeminjaman>asList(
                new AturanStatusMahasiswa(),
                new AturanMaksimalAktif(),
                new AturanTunggakanDenda(),
                new AturanBatasWaktu(),
                new AturanDurasiMaksimal(),
                new AturanKetersediaanStok(),
                new AturanBentrokJadwal(),
                new AturanJamOperasional(),
                new AturanKapasitasRuang()));
    }

    public void tambahAturan(AturanPeminjaman aturan) {
        daftarAturan.add(aturan);
    }

    /**
     * Seluruh pelanggaran dikumpulkan lebih dahulu agar pengguna
     * dapat memperbaiki banyak hal dalam satu kali perbaikan.
     */
    public void validasi(KonteksPengajuan konteks) throws AturanDilanggarException {
        List<String> pesan = new ArrayList<String>();
        for (AturanPeminjaman aturan : daftarAturan) {
            try {
                aturan.periksa(konteks);
            } catch (AturanDilanggarException e) {
                pesan.add(e.getMessage());
            }
        }
        if (!pesan.isEmpty()) {
            StringBuilder gabung = new StringBuilder();
            for (int i = 0; i < pesan.size(); i++) {
                if (i > 0) {
                    gabung.append("; ");
                }
                gabung.append(pesan.get(i));
            }
            throw new AturanDilanggarException("Validasi pengajuan", gabung.toString());
        }
    }

    public List<AturanPeminjaman> getDaftarAturan() {
        return Collections.unmodifiableList(daftarAturan);
    }
}

package sipinjam.layanan;

import java.time.LocalDate;
import java.time.temporal.ChronoUnit;

import sipinjam.model.DetailPeminjaman;
import sipinjam.model.KondisiBarang;
import sipinjam.model.Peminjaman;
import sipinjam.model.RincianDenda;

/** Perhitungan denda keterlambatan, kondisi barang, dan pembatalan ruangan. */
public class KalkulatorDenda {
    public static final double DENDA_PEMBATALAN = 50000.0;

    public RincianDenda hitung(Peminjaman peminjaman, LocalDate tanggalKembali) {
        long hari = peminjaman.hariTerlambat(tanggalKembali);
        double dendaTerlambat = 0.0;
        double dendaKondisi = 0.0;
        StringBuilder catatan = new StringBuilder();

        for (DetailPeminjaman baris : peminjaman.getDetail()) {
            double tarif = baris.getBarang().tarifHarian() * hari * baris.getJumlah();
            KondisiBarang akhir = baris.getKondisiAkhir() == null
                    ? KondisiBarang.BAIK : baris.getKondisiAkhir();
            double kondisi = baris.getBarang().biayaKondisi(akhir) * baris.getJumlah();
            baris.setSubtotalDenda(tarif + kondisi);
            dendaTerlambat += tarif;
            dendaKondisi += kondisi;
            if (kondisi > 0) {
                catatan.append(baris.getBarang().getNama()).append(": ").append(akhir.getLabel())
                        .append("; ");
            }
        }

        RincianDenda rincian = new RincianDenda(hari, dendaTerlambat, dendaKondisi);
        rincian.setKeterangan(catatan.length() == 0
                ? (hari > 0 ? "Terlambat " + hari + " hari" : "Tanpa denda")
                : catatan.toString());
        return rincian;
    }

    /** Denda pembatalan hanya berlaku bila dibatalkan kurang dari satu hari sebelum pemakaian. */
    public double dendaPembatalan(LocalDate tanggalPakai, LocalDate hariIni) {
        long selisih = ChronoUnit.DAYS.between(hariIni, tanggalPakai);
        return selisih < 1 ? DENDA_PEMBATALAN : 0.0;
    }
}

package sipinjam.layanan;

import java.time.LocalDate;
import java.time.LocalTime;
import java.util.ArrayList;
import java.util.List;

import sipinjam.model.BookingRuangan;
import sipinjam.model.JadwalRuangan;
import sipinjam.model.Ruangan;
import sipinjam.model.SlotWaktu;
import sipinjam.model.StatusBooking;
import sipinjam.repositori.RepositoriBooking;

/** Pengelolaan jadwal slot ruangan. */
public class LayananBookingRuangan {
    private static final LocalTime BUKA = LocalTime.of(7, 0);
    private static final LocalTime TUTUP = LocalTime.of(21, 0);

    private final RepositoriBooking repoBooking;

    public LayananBookingRuangan(RepositoriBooking repoBooking) {
        this.repoBooking = repoBooking;
    }

    /** Slot acuan satu jam mulai pukul 07.00 sampai 21.00. */
    public List<SlotWaktu> templateSlot() {
        List<SlotWaktu> hasil = new ArrayList<SlotWaktu>();
        for (int jam = BUKA.getHour(); jam < TUTUP.getHour(); jam++) {
            hasil.add(new SlotWaktu(jam, jam + 1));
        }
        return hasil;
    }

    public JadwalRuangan jadwal(Ruangan ruangan, LocalDate tanggal) {
        JadwalRuangan jadwal = new JadwalRuangan(ruangan, tanggal);
        for (BookingRuangan booking : repoBooking.padaTanggal(ruangan, tanggal)) {
            jadwal.tambah(booking);
        }
        return jadwal;
    }

    public List<SlotWaktu> slotKosong(Ruangan ruangan, LocalDate tanggal) {
        return jadwal(ruangan, tanggal).slotKosong(templateSlot());
    }

    public BookingRuangan cariBentrok(Ruangan ruangan, LocalDate tanggal, SlotWaktu slot) {
        return repoBooking.cariBentrok(ruangan, tanggal, slot);
    }

    /**
     * Memperbarui status booking mengikuti waktu berjalan:
     * disetujui menjadi terpakai saat slot dimulai, terpakai menjadi selesai saat slot berakhir.
     */
    public int perbaruiStatusBerjalan(LocalDate hariIni, LocalTime sekarang) {
        int berubah = 0;
        for (BookingRuangan booking : repoBooking.semua()) {
            if (booking.getStatus() == StatusBooking.DISETUJUI
                    && booking.getTanggal().equals(hariIni)
                    && !sekarang.isBefore(booking.getSlot().getJamMulai())
                    && sekarang.isBefore(booking.getSlot().getJamSelesai())) {
                booking.ubahStatus(StatusBooking.TERPAKAI);
                berubah++;
            } else if (booking.getStatus() == StatusBooking.TERPAKAI
                    && (booking.getTanggal().isBefore(hariIni)
                        || (booking.getTanggal().equals(hariIni)
                            && !sekarang.isBefore(booking.getSlot().getJamSelesai())))) {
                booking.ubahStatus(StatusBooking.SELESAI);
                berubah++;
            }
        }
        return berubah;
    }
}

/** Rekap peminjaman, keterlambatan, dan utilisasi ruangan. */
public class RekapPeminjaman {

    public Map<String, Integer> perKategori(List<Peminjaman> daftar) {
        Map<String, Integer> hasil = new LinkedHashMap<String, Integer>();
        for (Peminjaman peminjaman : daftar) {
            for (DetailPeminjaman baris : peminjaman.getDetail()) {
                String kategori = baris.getBarang().getKategori() == null
                        ? "Lainnya" : baris.getBarang().getKategori().getNama();
                Integer lama = hasil.get(kategori);
                hasil.put(kategori, (lama == null ? 0 : lama) + baris.getJumlah());
            }
        }
        return hasil;
    }

    public double tingkatKeterlambatan(List<Peminjaman> daftar) {
        int kembali = 0;
        int terlambat = 0;
        for (Peminjaman peminjaman : daftar) {
            if (peminjaman.getPengembalian() == null) {
                continue;
            }
            kembali++;
            if (peminjaman.hariTerlambat(peminjaman.getPengembalian().getTanggalKembali()) > 0) {
                terlambat++;
            }
        }
        return kembali == 0 ? 0.0 : (terlambat * 100.0 / kembali);
    }

    public Map<String, Integer> utilisasiPerJam(List<BookingRuangan> daftar) {
        Map<String, Integer> hasil = new LinkedHashMap<String, Integer>();
        for (BookingRuangan booking : daftar) {
            String jam = String.format("%02d:00", booking.getSlot().getJamMulai().getHour());
            Integer lama = hasil.get(jam);
            hasil.put(jam, (lama == null ? 0 : lama) + 1);
        }
        return hasil;
    }

    public int jamTerpakai(List<BookingRuangan> daftar) {
        int total = 0;
        for (BookingRuangan booking : daftar) {
            total += booking.getSlot().durasiJam();
        }
        return total;
    }

    public double totalDenda(List<Peminjaman> daftar) {
        double total = 0.0;
        for (Peminjaman peminjaman : daftar) {
            if (peminjaman.getPengembalian() != null) {
                total += peminjaman.getPengembalian().getTotalDenda();
            }
        }
        return total;
    }
}

/**
 * Induk seluruh sumber daya yang dapat dipinjam: barang inventaris dan ruangan.
 * Tarif denda ditentukan oleh masing-masing subclass (polymorphism).
 */
public abstract class Barang {
    private final String kode;
    private String nama;
    private KategoriBarang kategori;
    private int stok;
    private double hargaSatuan;
    private boolean perluPersetujuan;

    protected Barang(String kode, String nama, KategoriBarang kategori, int stok,
                     double hargaSatuan, boolean perluPersetujuan) {
        if (kode == null || kode.trim().isEmpty()) {
            throw new IllegalArgumentException("Kode barang tidak boleh kosong");
        }
        if (stok < 0) {
            throw new IllegalArgumentException("Stok tidak boleh negatif");
        }
        this.kode = kode.trim();
        this.nama = nama;
        this.kategori = kategori;
        this.stok = stok;
        this.hargaSatuan = hargaSatuan;
        this.perluPersetujuan = perluPersetujuan;
    }

    /** Nama jenis untuk ditampilkan dan disimpan pada berkas. */
    public abstract String getJenisBarang();

    /** Tarif denda keterlambatan per hari. */
    public abstract double tarifHarian();

    /** Biaya kerusakan atau kehilangan sesuai jenis barang. */
    public abstract double biayaKondisi(KondisiBarang kondisi);

    /**
     * Perhitungan denda yang sama untuk seluruh jenis barang,
     * memakai tarif yang berbeda dari subclass.
     */
    public double hitungDenda(long hariTerlambat, KondisiBarang kondisi) {
        long hari = Math.max(0, hariTerlambat);
        return (tarifHarian() * hari) + biayaKondisi(kondisi);
    }

    /** Ruangan tidak dipinjam per tanggal, melainkan per slot jam. */
    public boolean dikelolaPerSlot() {
        return false;
    }

    public String getKode() {
        return kode;
    }

    public String getNama() {
        return nama;
    }

    public void setNama(String nama) {
        if (nama == null || nama.trim().isEmpty()) {
            throw new IllegalArgumentException("Nama barang tidak boleh kosong");
        }
        this.nama = nama.trim();
    }

    public KategoriBarang getKategori() {
        return kategori;
    }

    public void setKategori(KategoriBarang kategori) {
        this.kategori = kategori;
    }

    public int getStok() {
        return stok;
    }

    public void setStok(int stok) {
        if (stok < 0) {
            throw new IllegalArgumentException("Stok tidak boleh negatif");
        }
        this.stok = stok;
    }

    public double getHargaSatuan() {
        return hargaSatuan;
    }

    public void setHargaSatuan(double hargaSatuan) {
        this.hargaSatuan = hargaSatuan;
    }

    public boolean isPerluPersetujuan() {
        return perluPersetujuan;
    }

    public void setPerluPersetujuan(boolean perluPersetujuan) {
        this.perluPersetujuan = perluPersetujuan;
    }

    @Override
    public String toString() {
        return nama + " [" + kode + "]";
    }
}

package sipinjam.model;

import java.time.LocalDate;

/** Pengajuan pemakaian ruangan pada satu slot jam. */
public class BookingRuangan {
    private final String id;
    private final Ruangan ruangan;
    private final Mahasiswa pemohon;
    private final LocalDate tanggalPengajuan;
    private LocalDate tanggal;
    private SlotWaktu slot;
    private int jumlahPeserta;
    private String keperluan;
    private StatusBooking status;
    private String catatanPetugas;
    private double dendaPembatalan;
    private boolean dendaLunas;

    public BookingRuangan(String id, Ruangan ruangan, Mahasiswa pemohon, LocalDate tanggalPengajuan,
                          LocalDate tanggal, SlotWaktu slot, int jumlahPeserta, String keperluan,
                          StatusBooking status, String catatanPetugas,
                          double dendaPembatalan, boolean dendaLunas) {
        if (ruangan == null || pemohon == null || tanggal == null || slot == null) {
            throw new IllegalArgumentException("Ruangan, pemohon, tanggal, dan slot wajib diisi");
        }
        this.id = id;
        this.ruangan = ruangan;
        this.pemohon = pemohon;
        this.tanggalPengajuan = tanggalPengajuan;
        this.tanggal = tanggal;
        this.slot = slot;
        this.jumlahPeserta = jumlahPeserta;
        this.keperluan = keperluan;
        this.status = status == null ? StatusBooking.DIAJUKAN : status;
        this.catatanPetugas = catatanPetugas == null ? "" : catatanPetugas;
        this.dendaPembatalan = dendaPembatalan;
        this.dendaLunas = dendaLunas;
    }

    public void ubahStatus(StatusBooking tujuan) {
        if (tujuan == null) {
            throw new IllegalArgumentException("Status tujuan tidak boleh kosong");
        }
        if (!status.bolehKe(tujuan)) {
            throw new IllegalStateException("Status " + status.getLabel()
                    + " tidak dapat berpindah ke " + tujuan.getLabel());
        }
        this.status = tujuan;
    }

    public boolean adaTunggakan() {
        return dendaPembatalan > 0 && !dendaLunas;
    }

    public String getId() {
        return id;

/** Kumpulan booking pada satu ruangan untuk satu tanggal. */
public class JadwalRuangan {
    private final Ruangan ruangan;
    private final LocalDate tanggal;
    private final List<BookingRuangan> daftar;

    public JadwalRuangan(Ruangan ruangan, LocalDate tanggal) {
        this.ruangan = ruangan;
        this.tanggal = tanggal;
        this.daftar = new ArrayList<BookingRuangan>();
    }

    public void tambah(BookingRuangan booking) {
        if (booking != null && booking.getTanggal().equals(tanggal)
                && booking.getRuangan().getKode().equals(ruangan.getKode())) {
            daftar.add(booking);
        }
    }

    /** Booking aktif yang bertumpang tindih dengan slot yang diminta. */
    public BookingRuangan cariBentrok(SlotWaktu slot) {
        for (BookingRuangan booking : daftar) {
            if (booking.getStatus().mengunciSlot() && booking.getSlot().bertumpangTindih(slot)) {
                return booking;
            }
        }
        return null;
    }

    /** Slot dari daftar acuan yang belum dipakai booking aktif. */
    public List<SlotWaktu> slotKosong(List<SlotWaktu> acuan) {
        List<SlotWaktu> hasil = new ArrayList<SlotWaktu>();
        for (SlotWaktu slot : acuan) {
            if (cariBentrok(slot) == null) {
                hasil.add(slot);
            }
        }
        return hasil;
    }

    public List<BookingRuangan> getDaftar() {
        return Collections.unmodifiableList(daftar);
    }

    public Ruangan getRuangan() {
        return ruangan;
    }

    public LocalDate getTanggal() {
        return tanggal;
    }
}

    public Peminjaman(String id, Mahasiswa pemohon, LocalDate tanggalPengajuan,
                      LocalDate tanggalMulai, LocalDate tanggalSelesai, String keperluan) {
        if (pemohon == null) {
            throw new IllegalArgumentException("Pemohon wajib diisi");
        }
        if (tanggalMulai == null || tanggalSelesai == null) {
            throw new IllegalArgumentException("Tanggal peminjaman wajib diisi");
        }
        if (tanggalSelesai.isBefore(tanggalMulai)) {
            throw new IllegalArgumentException("Tanggal selesai tidak boleh sebelum tanggal mulai");
        }
        this.id = id;
        this.pemohon = pemohon;
        this.tanggalPengajuan = tanggalPengajuan;
        this.tanggalMulai = tanggalMulai;
        this.tanggalSelesai = tanggalSelesai;
        this.keperluan = keperluan;
        this.detail = new ArrayList<DetailPeminjaman>();
        this.status = StatusPeminjaman.DIAJUKAN;
        this.catatanPetugas = "";
        this.idPembina = "";
        this.pengembalian = null;
    }

    public void tambahDetail(DetailPeminjaman baris) {
        if (baris == null) {
            throw new IllegalArgumentException("Detail peminjaman tidak boleh kosong");
        }
        for (DetailPeminjaman ada : detail) {
            if (ada.getBarang().getKode().equals(baris.getBarang().getKode())) {
                throw new IllegalArgumentException("Barang " + baris.getBarang().getNama()
                        + " sudah ada pada pengajuan ini");
            }
        }
        detail.add(baris);
    }

    public void ubahStatus(StatusPeminjaman tujuan) {
        if (tujuan == null) {
            throw new IllegalArgumentException("Status tujuan tidak boleh kosong");
        }
        if (!status.bolehKe(tujuan)) {
            throw new IllegalStateException("Status " + status.getLabel()
                    + " tidak dapat berpindah ke " + tujuan.getLabel());
        }
        this.status = tujuan;
    }

    public int durasiHari() {
        long hari = ChronoUnit.DAYS.between(tanggalMulai, tanggalSelesai);

    }

    public abstract Peran getPeran();

    public boolean cekPassword(String masukan) {

import java.time.Duration;

/**
 * Interval jam pemakaian ruangan.
 * Dua constructor disediakan agar objek dapat dibuat dari LocalTime
 * maupun dari jam bulat (overloading constructor).
 */
public class SlotWaktu {
    private final LocalTime jamMulai;
    private final LocalTime jamSelesai;

    public SlotWaktu(LocalTime jamMulai, LocalTime jamSelesai) {
        if (jamMulai == null || jamSelesai == null) {
            throw new IllegalArgumentException("Jam mulai dan jam selesai wajib diisi");
        }
        if (!jamSelesai.isAfter(jamMulai)) {
            throw new IllegalArgumentException("Jam selesai harus setelah jam mulai");
        }
        this.jamMulai = jamMulai;
        this.jamSelesai = jamSelesai;
    }

    public SlotWaktu(int jamMulai, int jamSelesai) {
        this(LocalTime.of(jamMulai, 0), LocalTime.of(jamSelesai, 0));
    }

    public LocalTime getJamMulai() {
        return jamMulai;
    }

    public LocalTime getJamSelesai() {
        return jamSelesai;
    }

    /** Lama pemakaian dalam satuan jam. */
    public int durasiJam() {
        return (int) Duration.between(jamMulai, jamSelesai).toMinutes() / 60;
    }

    /**
     * Bentrok bila mulai1 < selesai2 dan mulai2 < selesai1.
     * Slot yang berakhir tepat saat slot lain mulai tidak dianggap bentrok.
     */
    public boolean bertumpangTindih(SlotWaktu lain) {
        if (lain == null) {
            return false;
        }
        return jamMulai.isBefore(lain.jamSelesai) && lain.jamMulai.isBefore(jamSelesai);
    }

    public boolean kelipatanJam() {
        return jamMulai.getMinute() == 0 && jamSelesai.getMinute() == 0;
    }

    public boolean diDalam(LocalTime buka, LocalTime tutup) {
        return !jamMulai.isBefore(buka) && !jamSelesai.isAfter(tutup);
    }

    @Override
    public String toString() {
        return String.format("%02d:%02d-%02d:%02d", jamMulai.getHour(), jamMulai.getMinute(),
                jamSelesai.getHour(), jamSelesai.getMinute());
    }
}

    }

        if (barang == null) {
            throw new IllegalArgumentException("Barang tidak boleh kosong");

        if (cari(barang.getKode()) != null) {
            throw new IllegalArgumentException("Kode " + barang.getKode() + " sudah dipakai");

        for (Barang barang : daftar) {

            }

    }

        return barang instanceof Ruangan ? (Ruangan) barang : null;
    }

    public List<Barang> daftarBarang() {
        List<Barang> hasil = new ArrayList<Barang>();
        for (Barang barang : daftar) {

        return hasil;
    }

        List<Ruangan> hasil = new ArrayList<Ruangan>();
        for (Barang barang : daftar) {
            if (barang instanceof Ruangan) {
                hasil.add((Ruangan) barang);
            }
        }
        return hasil;
    }

    public List<Barang> semua() {
        return Collections.unmodifiableList(daftar);
    }

            }
        }
        return null;
    }

    /**
     * Jumlah unit barang yang sudah menahan stok pada rentang tanggal tertentu.
     * Hanya peminjaman berstatus DISETUJUI atau DIPINJAM yang dihitung.
     */
    public int terpakai(Barang barang, LocalDate mulai, LocalDate selesai) {

        for (Peminjaman peminjaman : daftar) {
            if (!peminjaman.getStatus().menahanStok()) {
                continue;

            if (!peminjaman.bertumpangTindih(mulai, selesai)) {
                continue;

            for (DetailPeminjaman baris : peminjaman.getDetail()) {
                if (baris.getBarang().getKode().equals(barang.getKode())) {
                    total += baris.getJumlah();
                }
            }
        }
        return total;
    }

        int aktif = 0;
        for (Peminjaman peminjaman : daftar) {
            if (peminjaman.getPemohon().getUsername().equals(mahasiswa.getUsername())
                    && peminjaman.getStatus().aktif()) {
                aktif++;
            }
        }

    }
